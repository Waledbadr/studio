const FIRESTORE_DISABLED = true;

// Flag to enable manual sync only
const ALLOW_MANUAL_SYNC = true;

export function AccommodationProvider({ children }: { children: ReactNode }) {
  // ... existing state ...
  
  const [firestoreDisabled] = useState(FIRESTORE_DISABLED);

  // ==================== EMERGENCY FIX ====================
  // Comment out ALL onSnapshot listeners
  // ======================================================

  /*
  // ❌ DISABLED: This was causing 12K reads per operation!
  useEffect(() => {
    if (!db || !auth?.currentUser) return;
    
    // Workers listener - DISABLED
    // const workersCol = collection(db, 'companies');
    // workersUnsubRef.current = onSnapshot(...);
    
    // Companies listener - DISABLED
    // const companiesCol = collection(db, 'companies');
    // companiesUnsubRef.current = onSnapshot(...);
    
    // ... etc
  }, [db, auth]);
  */

  // ==================== NEW APPROACH ====================
  // Load from localStorage ONLY, no Firestore
  // ======================================================

  useEffect(() => {
    console.log('🔒 [Emergency Mode] Firestore listeners DISABLED');
    console.log('💾 [Emergency Mode] Using localStorage ONLY');
    
    // Load from localStorage
    loadFromLocalStorage();
  }, []);

  function loadFromLocalStorage() {
    try {
      const w = localStorage.getItem('ac_workers');
      const o = localStorage.getItem('ac_occupants');
      const r = localStorage.getItem('estatecare_residences');
      const c = localStorage.getItem('ac_companies');
      const ct = localStorage.getItem('ac_contracts');
      const i = localStorage.getItem('ac_invoices');
      const h = localStorage.getItem('ac_history');

      if (w) setWorkers(JSON.parse(w));
      if (o) setOccupants(JSON.parse(o));
      if (r) {
        const parsed = JSON.parse(r);
        setResidences(parsed.map(mapComplexToResidence));
      }
      if (c) setCompanies(JSON.parse(c));
      if (ct) setContracts(JSON.parse(ct));
      if (i) setInvoices(JSON.parse(i));
      if (h) setAccommodationHistory(JSON.parse(h));

      console.log('✅ [Emergency Mode] Loaded from localStorage');
    } catch (e) {
      console.error('❌ [Emergency Mode] Failed to load from localStorage:', e);
    }
  }

  // ==================== MANUAL SYNC ====================
  // User can trigger sync manually when needed
  // ======================================================

  async function manualSync() {
    if (!ALLOW_MANUAL_SYNC || !db) {
      console.warn('⚠️ Manual sync not allowed or DB not available');
      return { ok: false, error: 'Sync not available' };
    }

    try {
      console.log('🔄 [Manual Sync] Starting...');
      
      // Sync with limits to minimize reads
      const [workersSnap, occupantsSnap] = await Promise.all([
        getDocs(query(collection(db, 'workers'), limit(200))),
        getDocs(query(collection(db, 'occupants'), limit(300))),
      ]);

      const newWorkers = workersSnap.docs.map(d => ({ id: d.id, ...d.data() }));
      const newOccupants = occupantsSnap.docs.map(d => ({ id: d.id, ...d.data() }));

      setWorkers(newWorkers as any);
      setOccupants(newOccupants as any);

      localStorage.setItem('ac_workers', JSON.stringify(newWorkers));
      localStorage.setItem('ac_occupants', JSON.stringify(newOccupants));

      console.log('✅ [Manual Sync] Complete:', {
        workers: newWorkers.length,
        occupants: newOccupants.length,
        reads: workersSnap.size + occupantsSnap.size,
      });

      toast({
        title: 'تم التحديث',
        description: `تم تحديث ${newWorkers.length} عامل و ${newOccupants.length} مسكن`,
      });

      return { ok: true, reads: workersSnap.size + occupantsSnap.size };
    } catch (e) {
      console.error('❌ [Manual Sync] Failed:', e);
      toast({
        title: 'فشل التحديث',
        description: 'حدث خطأ أثناء التحديث من قاعدة البيانات',
        variant: 'destructive',
      });
      return { ok: false, error: String(e) };
    }
  }

  // ==================== WRITE OPERATIONS ====================
  // Continue to write to Firestore, but don't listen
  // ========================================================

  // checkInWorker, checkOutWorker, etc. remain the same
  // They write to Firestore but don't trigger listeners

  return (
    <AccommodationContext.Provider
      value={{
        // ... existing values ...
        manualSync,  // NEW: Add manual sync function
        firestoreDisabled,  // NEW: Flag to show in UI
      }}
    >
      {children}
    </AccommodationContext.Provider>
  );
}
